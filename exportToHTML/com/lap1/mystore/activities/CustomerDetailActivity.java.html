    private void loadCustomerData() {
        SQLiteDatabase db = dbHelper.getReadableDatabase();

        Cursor cCust = db.rawQuery("SELECT name, phone FROM customers WHERE id = ?",
                new String[]{String.valueOf(customerId)});
        if (!cCust.moveToFirst()) { cCust.close(); finish(); return; }
        customerName = cCust.getString(0);
        customerPhone = cCust.getString(1);
        tvName.setText(customerName);
        tvPhone.setText("الهاتف: " + (customerPhone != null && !customerPhone.isEmpty() ? customerPhone : "غير متوفر"));
        cCust.close();

        currentBalance = dbHelper.getCustomerLedgerBalance(db, customerId);

        tvSummaryBalance.setText(String.format(Locale.US, "الرصيد المتبقي (الدين): %.2f ر.ي", currentBalance));

        Cursor cSummary = db.rawQuery(
                "SELECT COALESCE(SUM(CASE WHEN amount > 0 THEN amount ELSE 0 END),0), " +
                        "COALESCE(SUM(CASE WHEN amount < 0 THEN -amount ELSE 0 END),0) " +
                        "FROM account_ledger WHERE party_type = ? AND party_id = ?",
                new String[]{DatabaseHelper.PARTY_CUSTOMER, String.valueOf(customerId)});
        totalDebt = 0.0;
        totalPaid = 0.0;
        if (cSummary.moveToFirst()) { totalDebt = cSummary.getDouble(0); totalPaid = cSummary.getDouble(1); }
        cSummary.close();
        tvSummaryDebt.setText(String.format(Locale.US, "إجمالي الزيادات: %.2f ر.ي", totalDebt));
        tvSummaryPaid.setText(String.format(Locale.US, "إجمالي السداد: %.2f ر.ي", totalPaid));

        historyList.clear();
        Cursor cLedger = db.rawQuery(
                "SELECT al.id, al.transaction_type, al.reference_id, al.amount, al.transaction_date, al.notes, s.invoice_number, s.notes " +
                        "FROM account_ledger al LEFT JOIN sales s ON al.reference_id = s.id AND al.party_type = '" + DatabaseHelper.PARTY_CUSTOMER + "' " +
                        "WHERE al.party_type = ? AND al.party_id = ? ORDER BY al.transaction_date ASC, al.id ASC",
                new String[]{DatabaseHelper.PARTY_CUSTOMER, String.valueOf(customerId)});

        double runningBalance = 0.0;
        while (cLedger.moveToNext()) {
            long ledgerId = cLedger.getLong(0);
            String type = cLedger.getString(1);
            long refId = cLedger.isNull(2) ? -1 : cLedger.getLong(2);
            double amount = cLedger.getDouble(3);
            String date = cLedger.getString(4);
            String notes = cLedger.getString(5);
            String invoice = cLedger.getString(6);
            String saleNotes = cLedger.getString(7);
            if (notes == null || notes.isEmpty()) notes = saleNotes;
            if (invoice == null) invoice = "";

            runningBalance += amount;
            if (runningBalance < 0 && runningBalance > -0.01) runningBalance = 0;

            String title;
            if (DatabaseHelper.TX_OPENING.equals(type)) {
                title = "📌 رصيد افتتاحي";
            } else if (DatabaseHelper.TX_SALE.equals(type)) {
                title = "📄 فاتورة مبيعات #" + (invoice.isEmpty() ? refId : invoice);
            } else if (DatabaseHelper.TX_SALE_PAYMENT.equals(type)) {
                title = "💵 مدفوع عند الإصدار #" + (invoice.isEmpty() ? refId : invoice);
            } else if (DatabaseHelper.TX_PAYMENT.equals(type)) {
                title = "💵 سداد دين / قبض";
            } else if (DatabaseHelper.TX_RETURN.equals(type)) {
                title = "↩️ مرتجع من الفاتورة #" + (invoice.isEmpty() ? refId : invoice);
            } else {
                title = "🔹 حركة حساب";
            }

            String direction = amount > 0 ? "زيادة على الدين" : "تخفيض للدين";
            String entry = title +
                    "\nالتاريخ: " + (date == null ? "" : date) +
                    "\nالمبلغ: " + String.format(Locale.US, "%.2f ر.ي", Math.abs(amount)) + " — " + direction +
                    (notes != null && !notes.isEmpty() ? "\nملاحظات: " + notes : "") +
                    "\n📊 الرصيد بعد الحركة: " + String.format(Locale.US, "%.2f ر.ي", Math.max(0, runningBalance));
            historyList.add(entry);
        }
        cLedger.close();

        if (historyList.isEmpty()) historyList.add("لا توجد عمليات محاسبية مسجلة لهذا العميل حتى الآن.");
        historyAdapter.notifyDataSetChanged();

                    try {
                        double amount = Double.parseDouble(amountText);

                            Toast.makeText(this, "المبلغ أكبر من الدين الحالي للعميل", Toast.LENGTH_SHORT).show();
                            return;
                        }
                        processSettlement(spMethod.getSelectedItem().toString(), etNotes.getText().toString().trim(), amount);

                        Toast.makeText(this, "مبلغ السداد غير صالح", Toast.LENGTH_SHORT).show();
                    }
                })
                .setNegativeButton("إلغاء", null)
                .show();

    private void processSettlement(String paymentMethod, String notes, double forcedAmount) {
        double paymentAmount = forcedAmount;
        if (!Double.isFinite(paymentAmount) || paymentAmount <= 0) { Toast.makeText(this, "مبلغ السداد غير صالح", Toast.LENGTH_SHORT).show(); return; }

        SQLiteDatabase db = dbHelper.getWritableDatabase();
        double ledgerBalance = dbHelper.getCustomerLedgerBalance(db, customerId);
        if (paymentAmount > ledgerBalance + 0.01) {
            Toast.makeText(this, "مبلغ السداد أكبر من الدين الحالي للعميل!", Toast.LENGTH_SHORT).show();
            return;
        }

        db.beginTransaction();
        try {
            ContentValues cv = new ContentValues();
            cv.put("customer_id", customerId);
            cv.put("amount", paymentAmount);
            cv.put("payment_method", paymentMethod);
            cv.put("notes", notes == null || notes.isEmpty() ? "سداد دين من كشف الحساب" : notes);
            long paymentId = db.insertOrThrow("payments", null, cv);

            double remainingPayment = paymentAmount;
            Cursor cOutstanding = db.rawQuery(
                    "SELECT id, remaining FROM sales WHERE customer_id = ? AND remaining > 0.001 ORDER BY id ASC",
                    new String[]{String.valueOf(customerId)});
            while (cOutstanding.moveToNext() && remainingPayment > 0.001) {
                long saleId = cOutstanding.getLong(0);
                double invRemaining = cOutstanding.getDouble(1);
                double toApply = Math.min(remainingPayment, invRemaining);
                double newRemaining = Math.max(0.0, invRemaining - toApply);
                double salePaid = 0.0;
                Cursor cp = db.rawQuery("SELECT paid FROM sales WHERE id = ?", new String[]{String.valueOf(saleId)});
                if (cp.moveToFirst()) salePaid = cp.getDouble(0);
                cp.close();
                String newStatus = newRemaining <= 0.001 ? "مدفوعة بالكامل" : (salePaid + toApply > 0.001 ? "دفع جزئي" : "آجلة");

                db.execSQL("UPDATE sales SET remaining = ?, paid = paid + ?, payment_status = ? WHERE id = ?",
                        new Object[]{newRemaining, toApply, newStatus, saleId});
                ContentValues alloc = new ContentValues();
                alloc.put("payment_id", paymentId);
                alloc.put("sale_id", saleId);
                alloc.put("amount", toApply);
                db.insertOrThrow("customer_payment_allocations", null, alloc);
                remainingPayment -= toApply;
            }
            cOutstanding.close();

            dbHelper.recordCustomerTransaction(db, customerId, DatabaseHelper.TX_PAYMENT, paymentId, -paymentAmount, notes == null || notes.isEmpty() ? "سداد دين" : notes);
            dbHelper.syncCustomerBalance(db, customerId);

            db.setTransactionSuccessful();
            currentBalance = dbHelper.getCustomerLedgerBalance(db, customerId);
            Toast.makeText(this, "تم تسجيل السداد وتوزيعه على الفواتير بنجاح", Toast.LENGTH_LONG).show();
            loadCustomerData();
        } catch (Exception e) {
            Toast.makeText(this, "تعذر تسجيل السداد: " + e.getMessage(), Toast.LENGTH_LONG).show();
        } finally { db.endTransaction(); }
    }

                        float baseline,
                        Paint paint
                ) {

                    if (text == null) {
                        text = "";
                    }

                    canvas.drawText(
                            text,

                            paint

                    canvas.save();

                    canvas.translate(
                            rightX - width,
                            top
                    );

                    layout.draw(canvas);

                    canvas.restore();

                    return layout.getHeight();
                }

                /* 
                 * تقسيم النص إلى أسطر بدون فقدان النص. 
                 */
                java.util.ArrayList<String> wrap(
                        String text,
                        Paint paint,
                        float maxWidth
                ) {

                    java.util.ArrayList<String> result =
                            new java.util.ArrayList<>();

                    if (
                            text == null ||
                                    text.trim().isEmpty()
                    ) {

                        result.add("");

                        return result;
                    }

                    String[] paragraphs =
                            text.split(
                                    "\n",
                                    -1
                            );

                    for (
                            String paragraph :
                            paragraphs
                    ) {

                        if (
                                paragraph == null ||
                                        paragraph.trim().isEmpty()
                        ) {

                            result.add("");

                            continue;
                        }

                        String[] words =
                                paragraph.trim().split(
                                        "\\s+"
                                );

                        String currentLine =
                                "";

                        for (
                                String word :
                                words
                        ) {

                            if (word == null) {
                                continue;
                            }

                            if (
                                    currentLine.isEmpty()
                            ) {

                                currentLine =
                                        word;

                            } else {

                                String candidate =
                                        currentLine +
                                                " " +
                                                word;

                                if (
                                        paint.measureText(
                                                candidate
                                        ) <= maxWidth
                                ) {

                                    currentLine =
                                            candidate;

                                } else {

                                    result.add(
                                            currentLine
                                    );

                                    currentLine =
                                            word;
                                }
                            }
                        }

                        if (
                                !currentLine.isEmpty()
                        ) {

                            result.add(
                                    currentLine
                            );
                        }
                    }

                    return result;
                }

                        /* 
                         * البيان يأخذ عرضًا كبيرًا. 
                         * إذا طال، ينزل إلى أسطر إضافية. 
                         */
                        java.util.ArrayList<String>
                                descriptionLines =
                                h.wrap(
                                        row.description,
                                        descriptionPaint,
                                        205
                                );

                        int rowHeight =
                                Math.max(
                                        50,
                                        17 +
                                                (
                                                        descriptionLines.size()
                                                                * 12
                                                )
                                );

                        /* 
                         * إذا لم تكف المساحة، ننهي الصفحة. 
                         * لا نقص الحركة. 
                         */
                        if (
                                y + rowHeight >
                                        BOTTOM
                        ) {

                            Paint footerPaint =
                                    h.paint(
                                            Color.rgb(
                                                    125,
                                                    130,
                                                    138
                                            ),
                                            7,
                                            false
                                    );

                            h.drawCenter(
                                    canvas,
                                    "عمران هيبر للتموين والتجارة",
                                    PAGE_WIDTH / 2f,
                                    810,
                                    footerPaint
                            );

                            h.drawRight(
                                    canvas,
                                    "صفحة " +
                                            pageNumber,
                                    RIGHT,
                                    810,
                                    footerPaint
                            );

                            pdfDocument.finishPage(
                                    page
                            );

                            pageNumber++;

                            /* 
                             * نرجع مباشرة إلى بداية الحلقة 
                             * لإنشاء صفحة جديدة. 
                             */
                            continue;
                        }

                        /* 
                         * خلفية الصف بالتناوب 
                         */
                        int rowColor =
                                rowIndex % 2 == 0
                                        ? Color.WHITE
                                        : Color.rgb(
                                        249,
                                        250,
                                        252
                                );

                        Paint rowBackground =
                                h.paint(
                                        rowColor,
                                        1,
                                        false
                                );

                        canvas.drawRect(
                                LEFT,
                                y,
                                RIGHT,
                                y + rowHeight,
                                rowBackground
                        );

                        /* 
                         * خطوط الأعمدة 
                         */
                        Paint gridPaint =
                                new Paint(
                                        Paint.ANTI_ALIAS_FLAG
                                );

                        gridPaint.setColor(
                                Color.rgb(
                                        230,
                                        233,
                                        238
                                )
                        );

                        gridPaint.setStrokeWidth(
                                1
                        );

                        canvas.drawLine(
                                470,
                                y,
                                470,
                                y + rowHeight,
                                gridPaint
                        );

                        canvas.drawLine(
                                335,
                                y,
                                335,
                                y + rowHeight,
                                gridPaint
                        );

                        canvas.drawLine(
                                255,
                                y,
                                255,
                                y + rowHeight,
                                gridPaint
                        );

                        canvas.drawLine(
                                170,
                                y,
                                170,
                                y + rowHeight,
                                gridPaint
                        );

                        /* 
                         * ================================================= 
                         * التاريخ 
                         * ================================================= 
                         */

                        Paint datePaint =
                                h.paint(
                                        Color.rgb(
                                                70,
                                                76,
                                                85
                                        ),
                                        7.1f,
                                        false
                                );

                        java.util.ArrayList<String>
                                dateLines =
                                h.wrap(
                                        h.date(
                                                row.date
                                        ),
                                        datePaint,
                                        72
                                );

                        int dateY =
                                y + 17;

                        for (
                                String dateLine :
                                dateLines
                        ) {

                            h.drawCenter(
                                    canvas,
                                    dateLine,
                                    510,
                                    dateY,
                                    datePaint
                            );

                            dateY += 10;
                        }

                        /* 
                         * ================================================= 
                         * البيان 
                         * ================================================= 
                         */

                        int descY =
                                y + 15;

                        for (
                                String line :
                                descriptionLines
                        ) {

                            h.drawRight(
                                    canvas,
                                    line,
                                    458,
                                    descY,
                                    descriptionPaint
                            );

                            descY += 12;
                        }

                        /* 
                         * ================================================= 
                         * مدين 
                         * ================================================= 
                         */

                        Paint debitPaint =
                                h.paint(
                                        row.debit > 0.001
                                                ? Color.rgb(
                                                190,
                                                60,
                                                60
                                        )
                                                : Color.GRAY,
                                        7.8f,
                                        row.debit > 0.001
                                );

                        h.drawCenter(
                                canvas,
                                row.debit > 0.001
                                        ? h.money(
                                        row.debit
                                )
                                        : "-",
                                295,
                                y + 20,
                                debitPaint
                        );

                        /* 
                         * ================================================= 
                         * دائن 
                         * ================================================= 
                         */

                        Paint creditPaint =
                                h.paint(
                                        row.credit > 0.001
                                                ? Color.rgb(
                                                20,
                                                140,
                                                95
                                        )
                                                : Color.GRAY,
                                        7.8f,
                                        row.credit > 0.001
                                );

                        h.drawCenter(
                                canvas,
                                row.credit > 0.001
                                        ? h.money(
                                        row.credit
                                )
                                        : "-",
                                215,
                                y + 20,
                                creditPaint
                        );

                        /* 
                         * ================================================= 
                         * الرصيد 
                         * ================================================= 
                         */

                        Paint balancePaint =
                                h.paint(
                                        Color.rgb(
                                                45,
                                                52,
                                                62
                                        ),
                                        7.8f,
                                        true
                                );

                        h.drawCenter(
                                canvas,
                                h.money(
                                        row.balance
                                ),
                                112,
                                y + 20,
                                balancePaint
                        );

                        /* 
                         * الخط السفلي 
                         */
                        canvas.drawLine(
                                LEFT,
                                y + rowHeight,
                                RIGHT,
                                y + rowHeight,
                                gridPaint
                        );

                        y += rowHeight;

                        rowIndex++;
                    }

                    /* 
                     * ===================================================== 
                     * آخر الصفحة 
                     * ===================================================== 
                     */

                    boolean lastPage =
                            rowIndex >= rows.size();

                    if (
                            lastPage
                    ) {

                        int bottomY =
                                Math.max(
                                        y + 14,
                                        690
                                );

                        h.line(
                                canvas,
                                LEFT,
                                bottomY,
                                RIGHT,
                                bottomY,
                                Color.rgb(
                                        210,
                                        216,
                                        224
                                ),
                                1
                        );

                        bottomY += 24;

                        Paint finalPaint =
                                h.paint(
                                        finalBalance > 0.001
                                                ? Color.rgb(
                                                190,
                                                60,
                                                60
                                        )
                                                : Color.rgb(
                                                20,
                                                140,
                                                95
                                        ),
                                        9,
                                        true
                                );

                        h.drawRight(
                                canvas,
                                finalBalance > 0.001
                                        ? "باقي عليك من الحساب: " +
                                        h.money(
                                                finalBalance
                                        ) +
                                          " ر.ي"

                                        : "حالة الحساب: لا يوجد مبلغ مستحق حاليًا",

                                RIGHT,
                                bottomY,
                                finalPaint
                        );

                        bottomY += 23;

                        Paint notePaint =
                                h.paint(
                                        Color.rgb(
                                                110,
                                                117,
                                                126
                                        ),
                                        7.5f,
                                        false
                                );
                    }

                    /* 
                     * ===================================================== 
                     * Footer 
                     * ===================================================== 
                     */

                    h.line(
                            canvas,
                            LEFT,
                            790,
                            RIGHT,
                            790,
                            Color.rgb(
                                    220,
                                    224,
                                    230
                            ),
                            1
                    );

                    Paint footerPaint =
                            h.paint(
                                    Color.rgb(
                                            120,
                                            125,
                                            133
                                    ),
                                    7,
                                    false
                            );

                    h.drawRight(
                            canvas,
                            "عمران هيبر للتموين والتجارة",
                            RIGHT,
                            812,
                            footerPaint
                    );

                    h.drawCenter(
                            canvas,
                            "شكراً لتعاملكم معنا",
                            PAGE_WIDTH / 2f,
                            812,
                            footerPaint
                    );

                    h.drawRight(
                            canvas,
                            "صفحة " +
                                    pageNumber,
                            115,
                            812,
                            footerPaint
                    );

                    pdfDocument.finishPage(
                            page
                    );

                    if (
                            rowIndex >= rows.size()
                    ) {

                        break;
                    }

                    pageNumber++;
                }

                /* 
                 * ========================================================= 
                 * حفظ PDF 
                 * ========================================================= 
                 */

                FileOutputStream outputStream =
                        new FileOutputStream(
                                file
                        );

                pdfDocument.writeTo(
                        outputStream
                );

                outputStream.close();

                pdfDocument.close();

                /* 
                 * ========================================================= 
                 * مشاركة PDF 
                 * ========================================================= 
                 */

                Uri uri =
                        FileProvider.getUriForFile(
                                this,
                                getPackageName() +
                                        ".fileprovider",
                                file
                        );

                Intent intent =
                        new Intent(
                                Intent.ACTION_SEND
                        );

                intent.setType(
                        "application/pdf"
                );

                intent.putExtra(
                        Intent.EXTRA_STREAM,
                        uri
                );

                intent.putExtra(
                        Intent.EXTRA_SUBJECT,
                        "كشف حساب العميل - " +
                                customerName
                );

                intent.putExtra(
                        Intent.EXTRA_TEXT,
                        "كشف حساب العميل\n" +
                                customerName
                );

                intent.addFlags(
                        Intent.FLAG_GRANT_READ_URI_PERMISSION
                );

                runOnUiThread(() -> {

                    Toast.makeText(
                            this,
                            "تم تجهيز كشف الحساب بنجاح",
                            Toast.LENGTH_SHORT
                    ).show();

                    startActivity(
                            Intent.createChooser(
                                    intent,
                                    "مشاركة كشف الحساب"
                            )
                    );
                });

            } catch (Exception e) {

                try {
                    pdfDocument.close();
                } catch (Exception ignored) {
                }

                runOnUiThread(() -> {

                    Toast.makeText(
                            this,
                            "تعذر إنشاء كشف الحساب: " +
                                    e.getMessage(),
                            Toast.LENGTH_LONG
                    ).show();
                });
            }
        });
    }
}
